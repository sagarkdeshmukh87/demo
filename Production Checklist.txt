User List where Users are Active and not freezed
SELECT Id, Name, Username, Email, UserRole.Name, Profile.Name, LastLoginDate
FROM User
WHERE IsActive = TRUE
AND Id IN (
    SELECT UserId
    FROM UserLogin
    WHERE IsFrozen = FALSE
)
